#include <ESP8266WiFi.h>
#include <ESP8266WebServer.h>

// ---------- WIFI ----------
const char* ssid = "NOMBRE_DE_TU_WIFI";
const char* password = "CONTRASENA_DE_TU_WIFI";

// ---------- PINES ----------
const int PIN_RELE = 5;   // GPIO5 = D1 en NodeMCU
const int PIN_PIR  = 14;  // GPIO14 = D5
const int PIN_BOTON = 12; // GPIO12 = D6
const int PIN_LED = 4;    // GPIO4 = D2

// Cambiar a LOW si tu modulo de rele se activa
// cuando la entrada recibe LOW.
const int RELE_ON = HIGH;
const int RELE_OFF = LOW;

// ---------- SERVIDOR ----------
ESP8266WebServer server(80);

// ---------- ESTADOS ----------
bool armada = false;
bool movimiento = false;
bool alarmaDisparada = false;

// ---------- TEMPORIZADORES ----------
unsigned long ultimoCambioBoton = 0;
unsigned long ultimoCambioLed = 0;

const unsigned long debounceMs = 50;
const unsigned long intervaloLed = 250;

int lecturaAnterior = HIGH;
int estadoBoton = HIGH;
bool ledEncendido = false;

// ---------- FUNCIONES ----------
void actualizarSalidas() {
  // El rele solo activa la segunda placa
  // cuando la alarma se ha disparado.
  if (alarmaDisparada) {
    digitalWrite(PIN_RELE, RELE_ON);
  } else {
    digitalWrite(PIN_RELE, RELE_OFF);
  }
}

void cambiarEstadoAlarma(bool activar) {
  armada = activar;

  // Al desarmar, se apaga la alarma y se
  // borra el estado de movimiento.
  if (!armada) {
    alarmaDisparada = false;
    movimiento = false;
  }

  actualizarSalidas();
}

void comprobarBoton() {
  int lectura = digitalRead(PIN_BOTON);

  if (lectura != lecturaAnterior) {
    ultimoCambioBoton = millis();
    lecturaAnterior = lectura;
  }

  if (millis() - ultimoCambioBoton >= debounceMs) {
    if (lectura != estadoBoton) {
      estadoBoton = lectura;

      // INPUT_PULLUP: pulsado = LOW
      if (estadoBoton == LOW) {
        cambiarEstadoAlarma(!armada);
      }
    }
  }
}

void comprobarPIR() {
  movimiento = (digitalRead(PIN_PIR) == HIGH);

  if (armada && movimiento) {
    alarmaDisparada = true;
  }

  actualizarSalidas();
}

void controlarLED() {
  unsigned long ahora = millis();

  if (alarmaDisparada) {
    // Parpadeo rapido si se disparo la alarma
    if (ahora - ultimoCambioLed >= intervaloLed) {
      ultimoCambioLed = ahora;
      ledEncendido = !ledEncendido;
      digitalWrite(PIN_LED, ledEncendido);
    }
  } else if (armada) {
    // Luz fija cuando esta armada
    ledEncendido = true;
    digitalWrite(PIN_LED, HIGH);
  } else {
    // Apagado cuando esta desarmada
    ledEncendido = false;
    digitalWrite(PIN_LED, LOW);
  }
}

void manejarEstado() {
  String json = "{";
  json += "\"armada\":";
  json += (armada ? "true" : "false");
  json += ",\"movimiento\":";
  json += (movimiento ? "true" : "false");
  json += ",\"disparada\":";
  json += (alarmaDisparada ? "true" : "false");
  json += ",\"rele\":";
  json += (alarmaDisparada ? "true" : "false");
  json += "}";

  server.send(200, "application/json", json);
}

void manejarArmar() {
  cambiarEstadoAlarma(true);
  server.sendHeader("Location", "/");
  server.send(303);
}

void manejarDesarmar() {
  cambiarEstadoAlarma(false);
  server.sendHeader("Location", "/");
  server.send(303);
}

// ---------- CONFIGURACION ----------
void setup() {
  Serial.begin(115200);

  pinMode(PIN_RELE, OUTPUT);
  digitalWrite(PIN_RELE, RELE_OFF);

  pinMode(PIN_PIR, INPUT);
  pinMode(PIN_BOTON, INPUT_PULLUP);
  pinMode(PIN_LED, OUTPUT);
  digitalWrite(PIN_LED, LOW);

  Serial.println();
  Serial.println("Iniciando sistema de seguridad...");

  WiFi.mode(WIFI_STA);
  WiFi.begin(ssid, password);

  Serial.print("Conectando al WiFi");

  unsigned long inicioWiFi = millis();

  while (WiFi.status() != WL_CONNECTED &&
         millis() - inicioWiFi < 20000) {
    delay(250);
    Serial.print(".");
  }

  if (WiFi.status() == WL_CONNECTED) {
    Serial.println();
    Serial.print("Direccion IP: ");
    Serial.println(WiFi.localIP());
  } else {
    Serial.println();
    Serial.println("No se conecto al WiFi.");
    Serial.println("Se puede reintentar reiniciando la placa.");
  }

  server.on("/", []() {
    server.send(200, "text/html", crearPagina());
  });

  server.on("/estado", manejarEstado);
  server.on("/armar", manejarArmar);
  server.on("/desarmar", manejarDesarmar);

  server.begin();
  Serial.println("Servidor web iniciado");
}

// ---------- BUCLE PRINCIPAL ----------
void loop() {
  server.handleClient();

  comprobarBoton();
  comprobarPIR();
  controlarLED();

  yield();
}
